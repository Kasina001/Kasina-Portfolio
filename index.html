# Fine-tune a BERT classifier for sentiment
from transformers import AutoModelForSequenceClassification
import torch

@dataclass
class TrainConfig:
    model_name: str = "bert-base-uncased"
    num_labels: int = 3
    lr: float = 2e-5
    epochs: int = 4

def train(cfg: TrainConfig, loader):
    model = AutoModelForSequenceClassification.from_pretrained(
        cfg.model_name, num_labels=cfg.num_labels
    )
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=cfg.lr
    )
    for epoch in range(cfg.epochs):
        for batch in loader:
            loss = model(**batch).loss
            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
    return model
-- Churn risk score per customer segment
WITH activity AS (
  SELECT
    user_id,
    COUNT(*) AS sessions_30d,
    MAX(created_at) AS last_seen,
    SUM(revenue) AS ltv
  FROM events
  WHERE created_at >= NOW() - '30 days'::interval
  GROUP BY 1
),
scored AS (
  SELECT *,
    NTILE(10) OVER (
      ORDER BY sessions_30d ASC, ltv ASC
    ) AS churn_decile
  FROM activity
)
SELECT
  churn_decile,
  COUNT(*)       AS users,
  AVG(ltv)::int  AS avg_ltv,
  AVG(sessions_30d) AS avg_sessions
FROM scored
GROUP BY 1
ORDER BY 1;
// Animated D3 line chart in React
import { useEffect, useRef } from "react"
import * as d3 from "d3"

export function LineChart({ data, width=600, height=300 }) {
  const svgRef = useRef(null)

  useEffect(() => {
    const x = d3.scaleTime()
      .domain(d3.extent(data, d => d.date))
      .range([0, width])
    const y = d3.scaleLinear()
      .domain([0, d3.max(data, d => d.value)])
      .range([height, 0])
    const line = d3.line()
      .x(d => x(d.date))
      .y(d => y(d.value))
      .curve(d3.curveCatmullRom)

    d3.select(svgRef.current)
      .append("path")
      .datum(data).attr("d", line)
      .attr("stroke", "#00e5ff")
      .attr("fill", "none")
  }, [data])

  return <svg ref={svgRef} width={width} height={height} />
}
